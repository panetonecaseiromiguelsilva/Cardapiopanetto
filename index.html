        function downloadAsImage() {

            const btn = document.getElementById('btn-download-img');

            if (!btn) return;

            const originalHTML = btn.innerHTML;

            btn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i> Gerando imagem...';

            btn.disabled = true;

            const flyer = document.getElementById('flyer');

            setTimeout(() => {

                html2canvas(flyer, {

                    scale: 2,

                    useCORS: true,

                    allowTaint: true,

                    backgroundColor: null

                }).then(function(canvas) {

                    const link = document.createElement('a');

                    link.download = 'Panfleto_Panetones_Natal.png';

                    link.href = canvas.toDataURL('image/png');

                    link.click();

                    btn.innerHTML = '<i class="fa-solid fa-check"></i> Imagem Salva!';

                    setTimeout(() => {

                        btn.innerHTML = originalHTML;
